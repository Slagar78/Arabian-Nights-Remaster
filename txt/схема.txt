├── assets/                <-- ТОЛЬКО ДЛЯ СТАНДАРТНОГО GUI
│   ├── button.png         <-- Кнопки "Меню" и "Выход"

├── assets_game/           <-- ТОЛЬКО ДЛЯ ИГРЫ
│   ├── characters/

│   │       ├── Shukran_down.png

│   ├── portraits/         <-- HD-портреты для диалогов
